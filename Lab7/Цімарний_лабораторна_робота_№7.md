## 3 Отримані результати
<h3 align="center">Завдання №1</h3>

<span style="font-size: 18px">**Скрипт**</span>
```m
clc

while true
  n = inputdlg({'x =', 'Кількість повторів N=', 'Відповідь:'}, 'Варіант 16', 1);
  if isempty(n)
    errordlg('Введіть значення x, N!!!', 'Помилка');
    ans_user = input('Bazhaete prodovzhiti?(Y/N) ', 's');
      if ~strcmpi(ans_user, 'Y')
        disp('Prohrama zavershena.');
        break;
      else
        continue;
      end
  end

  x = str2double(n{1});
  N = str2double(n{2});

  has_div_by_zero = false;
  if ~isnan(x) && ~isnan(N)
    for k = 0:N
      if x + k == 0
        has_div_by_zero = true;
        break;
      end
    end
  end

  if isnan(x) || isnan(N) || x == 0 || N <= 0 || fix(N) ~= N || has_div_by_zero
    errordlg('Введіть коректні числа (x != 0, N > 0)!', 'Помилка');
    ans_user = input('Bazhaete prodovzhiti?(Y/N) ', 's');
      if ~strcmpi(ans_user, 'Y')
        disp('Prohrama zavershena.');
        break;
      else
        continue;
      end
  end

  % Початкове значення першого доданка (1/x) та суми
  term = 1 / x;
  S = term;

  for k = 1:N
    term = term / (x + k); % Рекурентне обчислення наступного доданка
    S = S + term; % Додавання до загальної суми
  end

  x_str = num2str(x);
  N_str = num2str(N);
  S_str = num2str(S, '%.8g');
  inputdlg({'x =', 'Кількість повторів N=', 'Відповідь:'}, ...
            'Варіант 16', 1, {x_str, N_str, S_str});

  ans_user = input('Bazhaete prodovzhiti?(Y/N) ', 's');
  if ~strcmpi(ans_user, 'Y')
    disp('Prohrama zavershena.');
    break;
  end
end
```

```m
clc
c = 1.7;
i = 0;

while true
  m = input("\nVvedit vektor stovpchik [n1; n2; ...]:\n");
  elem = [];

  if ~isnumeric(m) || size(m, 2) > 1 || isempty(m) || ~isreal(m)
    errordlg('Введіть числовий вектор стовпчик!!!', 'Помилка');
    s = input('Bazhaete prodovzhiti? (y/n): ', 's');
    if ~strcmpi(s, 'Y')
      disp('Prohrama zavershena.');
      break;
    else
      continue;
    end
  end

  i = i + 1;
  fprintf('\nRozrahunok %d\n', i);

  for k = 1:size(m, 1)
    if m(k, 1) >= c
      elem = [elem, m(k, 1)];
    end
  end

  n = numel(elem);

  if isempty(elem)
    fprintf('Elementiv >= %g ne znaydeno\n', c);
  elseif n == 1
    fprintf('Kilkist elementiv >= %g: %g\n\nElement odin, dobutku nemaye\n', ...
         c, n);
  else
    fprintf('Kilkist elementiv >= %g: %g\n\nDobutok elementiv: %g\n', ...
         c, n, prod(elem));
  end

  ans_user = input('Bazhaete prodovzhiti?(Y/N) ', 's');
  if ~strcmpi(ans_user, 'Y')
    disp('Prohrama zavershena.');
    break;
  end
end
```

<h3 align="center">Завдання №3</h1>

<span style="font-size: 18px">**Скрипт**</span>
```m
clc
i = 0;

while true
  a = str2double(input('Vvedit a = ', 's'));
  b = str2double(input('Vvedit b = ', 's'));

  if isnan(a) || isnan(b) || a >= b
    errordlg('Введено невірні значення!!!', 'Помилка');
    s = input('Bazhaete prodovzhiti? (y/n): ', 's');
    if ~strcmpi(s, 'Y')
      disp('Prohrama zavershena.');
      break;
    else
      continue;
    end
  end

  i = i + 1;
  fprintf('\nRozrahunok %d\n', i);

  h = 0.1 * (b - a);
  x = a:h:b;
  y = exp(-2 * x);
  elem = [];

  for k = 1:size(y, 2)
    if y(1, k) > 4.1
      elem = [elem, y(1, k)];
    end
  end

  n = numel(elem);

  if isempty(elem)
    fprintf('Elementiv > 4.1 ne znaydeno\n');
  elseif n == 1
    fprintf('Kilkist elementiv > 4.1: %g\n\nElement odin, dobutku nemaye\n', n);
  else
    fprintf('Kilkist elementiv > 4.1: %g\n\nDobutok elementiv: %g\n', n, prod(elem));
  end

  ans_user = input('Bazhaete prodovzhiti?(Y/N) ', 's');
  if ~strcmpi(ans_user, 'Y')
    disp('Prohrama zavershena.');
    break;
  end
end
```

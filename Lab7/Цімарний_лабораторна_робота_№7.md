<h3 align="center">Завдання №1</h3>

<span style="font-size: 18px">**Скрипт**</span>
```m
clc
n = inputdlg({'x =', 'Кількість повторів N=', 'Відповідь:'}, 'Варіант 16', 1);
x = str2double(n(1));
N = str2double(n(2));

% Початкове значення першого доданка (1/x) та суми
n = 1 / x;
S = n;

for k = 1:N
  n = n / (x + k); % Рекурентне обчислення наступного доданка
  S = S + n; % Додавання до загальної суми
end

x_str = num2str(x);
N_str = num2str(N);
S_str = num2str(S);
n = inputdlg({'x =', 'Кількість повторів N=', 'Відповідь:'}, ...
              'Варіант 16', 1, {x_str, N_str, S_str});
```

<h3 align="center">Завдання №2</h1>

<span style="font-size: 18px">**Скрипт**</span>
```m
clc
m = input("Vvedit vektor stovpchik [n1; n2; ...]:\n ");
elem = [];
c = 1.7;

if ~isnumeric(m) || size(m, 2) > 1 || isempty(m)
  errordlg('Введіть числовий вектор стовпчик!!!', 'Помилка');
  s = input('Bazhaete prodovzhiti? (y/n): ', 's');
  if ~strcmpi(s, 'Y')
    disp('Prohrama zavershena.');
    return
  else
    Lab7_2
    return
  end
else
  for k = 1:size(m, 1)
    if m(k, 1) >= c
      elem = [elem, m(k, 1)];
    end
  end
end

n = numel(elem);

if isempty(elem)
  fprintf('Elementiv >= %g ne znaydeno\n', c);
elseif n == 1
  fprintf('Kilkist elementiv >= %g: %g\n\nElement odin, dobutku nemaye\n', ...
       c, n);
else
  fprintf('Kilkist elementiv >= %g: %g\n\nDobutok elementiv: %g\n', ...
       c, n, prod(elem));
end
```

<h3 align="center">Завдання №3</h1>

<span style="font-size: 18px">**Скрипт**</span>
```m
clc
a = input('Vvedit a = ');
b = input('Vvedit b = ');

if isempty(a) || isempty(b) || ~isnumeric(a) || ~isnumeric(b) || a >= b
  errordlg('Значення "a" має бути меншим за "b" i обидва мають бути числами!', 'Помилка');
  s = input('Bazhaete prodovzhiti? (y/n): ', 's');
  if ~strcmpi(s, 'Y')
    disp('Prohrama zavershena.');
    return
  else
    Lab7_3
    return
  end
end

h = 0.1 * (b - a);
x = a:h:b;
y = exp(-2 * x);
elem = [];

for k = 1:size(y, 2)
  if y(1, k) > 4.1
    elem = [elem, y(1, k)];
  end
end

n = numel(elem);

if isempty(elem)
  fprintf('Elementiv > 4.1 ne znaydeno\n');
elseif n == 1
  fprintf('Kilkist elementiv > 4.1: %g\n\nElement odin, dobutku nemaye\n', n);
else
  fprintf('Kilkist elementiv > 4.1: %g\n\nDobutok elementiv: %g\n', n, prod(elem));
end
```
